--02_Analysis(Monthly Revenue trend)
SELECT order_date, strftime('%Y-%m', order_date)
FROM sales
LIMIT 10;

--High value customer 
SELECT
	customer_id,
	customer_name,
	sum(total_price) AS total_spent,
	count(order_id) AS total_orders
	FROM sales
	GROUP BY customer_id,
	customer_name
	ORDER BY total_spent DESC
	LIMIT 10;
